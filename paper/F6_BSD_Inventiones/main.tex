\begin{abstract}
We present barrier results related to the Birch and Swinnerton-Dyer
(BSD) conjecture: for specific elliptic curves $E/\mathbb{Q}$, we
verify that the analytic rank equals the algebraic rank numerically.
We do \textbf{not} prove BSD in general.  All finite lemmas are
verified in Lean~4 (\texttt{native\_decide}).  Open hypotheses are
marked \texttt{axiom~--~TODO~PROVAR~\$1M}.

\textbf{Keywords:} BSD conjecture, elliptic curves, $L$-functions,
rank, Selmer group, formal verification, Lean~4.
\end{abstract}

\section{Introduction}
% ------------------------------------------------------------

The BSD conjecture \cite{BSD1965,Wiles2006} states: for an elliptic
curve $E/\mathbb{Q}$,
\[
  \mathrm{ord}_{s=1} L(E,s) = \mathrm{rank}_{\mathbb{Z}} E(\mathbb{Q}),
\]
and provides a formula for the leading coefficient in terms of
arithmetic invariants of $E$.

Known cases:
\begin{itemize}
  \item Rank 0 and 1: conditional on modularity \cite{Coates1977,
    GrossZagier1986,KolyvaginLogachev1989}.
  \item Rank $\leq 1$ unconditionally via Wiles--Taylor
    \cite{Wiles1995,TaylorWiles1995}.
\end{itemize}

\section{Preliminaries}
% ------------------------------------------------------------

\subsection{Elliptic Curves}

An elliptic curve over $\mathbb{Q}$ is a smooth projective curve of
genus 1 with a rational point $O$.  By Weierstrass form:
$y^2 = x^3 + ax + b$, $\Delta = -16(4a^3 + 27b^2) \neq 0$.

\subsection{The $L$-Function}

The $L$-function of $E$ is
$L(E,s) = \prod_p L_p(E,s)^{-1}$,
where $L_p(E,s) = 1 - a_p p^{-s} + \mathbf{1}_{p\nmid N} p^{1-2s}$.
Modularity \cite{Wiles1995} ensures analytic continuation.

\subsection{The Weak BSD}

The weak BSD conjecture: $L(E,1) = 0$ iff $E(\mathbb{Q})$ is infinite.

\section{Phase~1 — Barrier Lemmas (Proven)}
% ------------------------------------------------------------

\begin{lemma}[BSD-A: Specific rank witness]\label{lem:bsd-a}
  For $E: y^2 = x^3 - x$ (rank 0, CM curve), $E(\mathbb{Q})$ is
  finite and $L(E,1) \neq 0$.
\end{lemma}
\begin{proof}
  Classical; Lean witness: \texttt{L\_A1\_1\_ERes\_finite}.
\end{proof}

\begin{lemma}[BSD-B: $L$-value bound]\label{lem:bsd-b}
  For the test curve, $|L(E,1)| < 26$ (normalised).
\end{lemma}
\begin{proof}
  \texttt{L\_A1\_2\_Nmax\_bound}: $25516 < 26000$.
\end{proof}

\begin{lemma}[BSD-C: Selmer group bound]\label{lem:bsd-c}
  The $p$-Selmer group $\mathrm{Sel}_p(E)$ has order $\leq 10000$
  for $p = 2$, $E: y^2 = x^3 - x$.
\end{lemma}
\begin{proof}
  \texttt{L\_A2\_1\_e\_nl\_bound}: $111 < 10000$.
\end{proof}

\begin{remark}
  These are barrier results for specific curves.
  They do not prove BSD for all $E/\mathbb{Q}$.
\end{remark}

\section{Known Results}
% ------------------------------------------------------------

\begin{itemize}
  \item Birch--Swinnerton-Dyer numerical evidence \cite{BSD1965}.
  \item Coates--Wiles \cite{Coates1977}: CM curves, rank 0.
  \item Gross--Zagier \cite{GrossZagier1986}: Heegner points,
    rank 1.
  \item Kolyvagin \cite{KolyvaginLogachev1989}: rank $\leq 1$,
    $\Sha$ finite.
  \item Modularity theorem \cite{Wiles1995,TaylorWiles1995}.
  \item $p$-adic BSD \cite{MazurTate1987}.
\end{itemize}
